def normalize_text(text):
    if not text:
        return ""
    return "".join(text.lower().replace("_", "").split())

from docx.shared import Pt

def replace_in_paragraphs(doc, field_values):
    for para in doc.paragraphs:
        if not para.runs:
            continue

        # 🔹 Merge all runs into one string
        full_text = "".join(run.text for run in para.runs)
        updated_text = full_text

        for field, value in field_values.items():
            
            # 1️⃣ Handle {{Field Name}} placeholders (case-insensitive)
            placeholder_pattern = re.compile(
                r"\{\{\s*" + re.escape(field) + r"\s*\}\}",
                re.IGNORECASE
            )
            updated_text = placeholder_pattern.sub(value, updated_text)

            # 2️⃣ Handle plain text field names (case-insensitive)
            field_pattern = re.compile(
                re.escape(field),
                re.IGNORECASE
            )
            updated_text = field_pattern.sub(value, updated_text)

        # 🔹 If no change, skip
        if updated_text == full_text:
            continue

        # 🔹 Preserve paragraph-level formatting
        first_run = para.runs[0]
        font_name = first_run.font.name or "Arial"
        font_size = first_run.font.size or Pt(11)
        bold = first_run.bold
        italic = first_run.italic
        underline = first_run.underline
        color = first_run.font.color.rgb

        # 🔹 Rewrite paragraph
        para.clear()
        run = para.add_run(updated_text)

        run.font.name = font_name
        run.font.size = font_size
        run.bold = bold
        run.italic = italic
        run.underline = underline
        if color:
            run.font.color.rgb = color

from docx.shared import Pt

def replace_in_tables(doc, field_values):
    for table in doc.tables:
        for row in table.rows:
            if len(row.cells) < 2:
                continue

            left_cell = row.cells[0]
            right_cell = row.cells[1]

            left_text = left_cell.text.strip()

            for field, value in field_values.items():
                if normalize_text(left_text) != normalize_text(field):
                    continue

                # 🔹 Get font from left cell
                font_name = "Arial"
                font_size = Pt(11)
                bold = None
                italic = None
                underline = None
                color = None

                if left_cell.paragraphs and left_cell.paragraphs[0].runs:
                    src_run = left_cell.paragraphs[0].runs[0]
                    font_name = src_run.font.name or font_name
                    font_size = src_run.font.size or font_size
                    bold = src_run.bold
                    italic = src_run.italic
                    underline = src_run.underline
                    color = src_run.font.color.rgb

                # 🔹 Clear right cell and write value with same formatting
                right_cell.text = ""
                p = right_cell.paragraphs[0]
                run = p.add_run(value)

                run.font.name = font_name
                run.font.size = font_size
                run.bold = bold
                run.italic = italic
                run.underline = underline
                if color:
                    run.font.color.rgb = color

import pikepdf

def apply_pdf_password(pdf_path, password):
    with pikepdf.open(pdf_path,allow_overwriting_input=True) as pdf:
        pdf.save(
            pdf_path,
            encryption=pikepdf.Encryption(
                user=str(password),
                owner=str(password)
            )
        )

def convert_word_to_pdf_with_password(word_path, pdf_path, password):
    word = win32com.client.DispatchEx("Word.Application")
    word.Visible = False

    try:
        doc = word.Documents.Open(word_path)

        # Save as PDF with password
        doc.ExportAsFixedFormat(
            OutputFileName=pdf_path,
            ExportFormat=17,   # wdExportFormatPDF
        )

        doc.Close(False)
    finally:
        try:
            doc.Close(False)
        except:
            pass
        try:
            word.Quit()
        except:
            pass
        
        doc = None
        word = None
        gc.collect()
        time.sleep(1)
        
    if password:
        apply_pdf_password(pdf_path,password)
    

def insert_signatory_image(doc, image_path):
    """
    If image_path exists → insert image.
    If no image_path → remove the 'signatory' text from template.
    """

    for para in doc.paragraphs:
        if "signatory" in para.text.lower():

            # 🔹 If image provided → insert image
            if image_path and os.path.exists(image_path):
                para.clear()
                run = para.add_run()
                run.add_picture(image_path, width=Inches(1.8))

            # 🔹 If NO image → remove text
            else:
                para.clear()

            break

def validate_date_format(date_str):
    """
    Validates DD-MM-YYYY format strictly
    """
    try:
        datetime.strptime(date_str, "%d-%m-%Y")
        return True
    except ValueError:
        return False

        self.region_cb["values"] = sorted(hro_df[REGION_COL].dropna().unique())

        self.entity_cb["values"] = sorted(
            hro_df[hro_df[REGION_COL] == self.region_var.get()][ENTITY_COL].dropna().unique()
        )

    def on_entity_change(self, _):
        self.category_cb["values"] = sorted(
            hro_df[
                (hro_df[REGION_COL] == self.region_var.get()) &
                (hro_df[ENTITY_COL] == self.entity_var.get())
            ][CATEGORY_COL].dropna().unique()
        )

    def on_category_change(self, _):
        self.letter_cb["values"] = sorted(
            hro_df[
                (hro_df[REGION_COL] == self.region_var.get()) &
                (hro_df[ENTITY_COL] == self.entity_var.get()) &

        )

        def upload_bulk_template():
            file_path = filedialog.askopenfilename(
                title="Upload Filled Bulk Template",
                filetypes=[("Excel Files", "*.xlsx")]
            )

            if not file_path:
                return

            try:
                bulk_df = pd.read_excel(file_path)
            except Exception as e:
                messagebox.showerror("Error", f"Failed to read file:\n{str(e)}")
                return

            # ==============================
            # 🔥 BUILD EXPECTED COLUMNS
            # ==============================
            expected_columns = []

            # Mandatory fields
            mandatory_fields = [ "Employee ID", "Manager Name", OUTPUT_FOLDER_COL, FOLDER_PASSWORD_COL]
            for field in mandatory_fields:
                if field not in expected_columns:
                    expected_columns.append(field)

            # Manual fields
            for col in DYNAMIC_FIELDS:
                hro_value = self.selected_hro_row[col]

                if pd.isna(hro_value):
                    continue

                hro_value_clean = str(hro_value).strip().upper()
                col_normalized = col.strip().lower().replace("_", " ")

                if hro_value_clean in ["NA", "WORKDAY"]:
                    continue

                if col_normalized in ["signatory", "letter effective date"]:
                    continue

                if "MANUAL" in hro_value_clean:
                    expected_columns.append(col)

            # ==============================
            # 🔥 VALIDATION
            # ==============================
            def normalize(col):
                return str(col).strip().lower().replace("_", "").replace(" ", "")

            normalized_expected = [normalize(c) for c in expected_columns]
            normalized_uploaded = [normalize(c) for c in bulk_df.columns]

            print("EXPECTED:", expected_columns)
            print("UPLOADED:", list(bulk_df.columns))

            if normalized_expected != normalized_uploaded:

                missing = [
                    expected_columns[i]
                    for i, col in enumerate(normalized_expected)
                    if col not in normalized_uploaded
                ]

                extra = [
                    bulk_df.columns[i]
                    for i, col in enumerate(normalized_uploaded)
                    if col not in normalized_expected
                ]

                error_msg = "Invalid template uploaded.\n\n"

                if missing:

    def download_bulk_template(self):
        if self.selected_hro_row is None:
            messagebox.showerror("Error", "No letter template selected.")
            return

        bulk_fields = []

        # ✅ Mandatory identity fields
        mandatory_fields = [ "Employee ID", "Manager Name", OUTPUT_FOLDER_COL, FOLDER_PASSWORD_COL]
        for field in mandatory_fields:
                if field not in bulk_fields:
                    bulk_fields.append(field)

        # ✅ ONLY MANUAL fields
        for col in DYNAMIC_FIELDS:
            hro_value = self.selected_hro_row[col]

            if pd.isna(hro_value):
                continue

            hro_value_clean = str(hro_value).strip().upper()
            col_normalized = col.strip().lower().replace("_", " ")

            # ❌ Exclusions
            if hro_value_clean in ["NA", "WORKDAY"]:
                continue

            if col_normalized in ["signatory", "letter effective date"]:
                continue

            # ✅ INCLUDE ONLY MANUAL
            if "MANUAL" in hro_value_clean:
                if col not in bulk_fields:
                    bulk_fields.append(col)

        if len(bulk_fields) == len(mandatory_fields):
            messagebox.showerror(
                "Error",
                "No MANUAL fields found for this letter."
            )
            return

        bulk_df = pd.DataFrame(columns=bulk_fields)

        save_path = filedialog.asksaveasfilename(
            defaultextension=".xlsx",
            filetypes=[("Excel Files", "*.xlsx")],
            title="Save Bulk Input Template As"
        )

        if not save_path:
            return

        bulk_df.to_excel(save_path, index=False)

        messagebox.showinfo(
            "Success",
            f"Bulk template downloaded successfully:\n\n{save_path}"
        )

    def validate_workday_fields_for_employee(self, emp_id, employee_row):
        """
        Returns:
            (True, field_values_dict)  -> if validation passes
            (False, error_message)     -> if validation fails
        """

        field_values = {}

        # Loop ONLY through fields required by selected letter
        for col in DYNAMIC_FIELDS:
            hro_value = self.selected_hro_row[col]

            # Only WORKDAY fields
            if pd.isna(hro_value) or str(hro_value).strip().upper() != "WORKDAY":
                continue

            # 🔹 Case 1: Column missing from Workday file
            if col not in letter_input_df.columns:
                return (
                    False,
                    f"{col} column is missing from Workday file"
                )

            value = employee_row[col]

            # 🔹 Case 2: Value missing for this employee
            if pd.isna(value) or str(value).strip() == "":
                return (
                    False,
                    f"{col} value is missing in Workday file for Employee ID {emp_id}"
                )

            # ✅ Valid value
            field_values[col] = str(value)

        return True, field_values

        zip_files = {}
        
        

        total = len(bulk_df)
        completed = 0
        errors = []
        success_count = 0
        temp_files = []
        
        # =========================
        # PROGRESS TRACKING
        # =========================

        total_employees = len(bulk_df)

        # Phase weights
        # 50% = document creation
        # 40% = PDF conversion
        # 10% = zip finalization

        creation_weight = 50
        pdf_weight = 40
        final_weight = 10
        
        uncategorized_exists = False
        missing_employee_rows = []

        for _, row in bulk_df.iterrows():

            folder_name_check = str(

                row.get("Manager Name", "")

            # ==========================

            folder_name = str(
                row.get(OUTPUT_FOLDER_COL, "")
            ).strip()

            folder_password = str(
                row.get(FOLDER_PASSWORD_COL, "")

            print("Row:", idx)
            print("Raw:",repr(row.get(FOLDER_PASSWORD_COL)))
            print("after:",repr(folder_password))
            print("-"*40)

            if folder_name.lower() == "nan":

            if folder_password.lower() == "nan":
                folder_password = ""

            folder_password = folder_password.strip()

                folder_password = ""

                       
            # 🔹 Find employee in Workday
            match = letter_input_df[
                letter_input_df[EMP_ID_COL].astype(str).str.strip() == emp_id
            ]

            

            if match.empty:
                errors.append(f"{emp_id} – Employee ID not found in Workday file")
                continue

            employee_row = match.iloc[0]

            # ==========================
            # UNCATEGORIZED / PASSWORD LOGIC
            # ==========================

            emp_override = None

            if uncategorized_override:
                emp_override = uncategorized_override["employee_data"].get(emp_id)
            print(uncategorized_override["employee_data"].keys())

            # User supplied details from popup
            if emp_override:
                folder_name = emp_override["folder_name"]

                if is_pdf:
                    folder_password = emp_override["password"]

            # Default folder if blank
            folder_name = "" if pd.isna(folder_name) else str(folder_name).strip()

            if not folder_name:
                folder_name = "Uncategorized"

            print("Employee:", emp_id)
            print("Password before IF:", repr(folder_password))
            # =====================================================
            # SYSTEM GENERATED PASSWORD
            # Use ONLY when Folder Password is blank
            # =====================================================
            if is_pdf and folder_password == "":

                first_name = str(
                    employee_row["Forename"]
                ).strip().replace(" ", "").upper()

                first_four = (
                    first_name[:4]
                    if len(first_name) >= 4
                    else first_name.ljust(4, "X")
                )

                
                dob = employee_row[" Date of Birth"]

                if pd.notna(dob):
                    if isinstance(dob, (pd.Timestamp, datetime)):
                        birth_year = dob.strftime("%Y")
                    else:
                        birth_year = pd.to_datetime(dob).strftime("%Y")
                else:
                    raise ValueError(
                        f"Date of Birth missing in Workday Input.xlsx for Employee ID {emp_id}"
                    )

                folder_password = first_four + birth_year

            folder_name = "" if pd.isna(folder_name) else str(folder_name).strip()
            if folder_name == "":
                folder_name = "Uncategorized"
            # Create ZIP after final folder name is decided
            if folder_name not in zip_files:

                zip_path = os.path.join(
                    manager_folder,
                    f"{folder_name}.zip"
                )

                zip_files[folder_name] = {
                    "zip_path": zip_path,
                    "zip_obj": zipfile.ZipFile(
                        zip_path,
                        "w",
                        zipfile.ZIP_DEFLATED
                    )
                }



            # 🔹 Validate required WORKDAY fields
            is_valid, result = self.validate_workday_fields_for_employee(
                emp_id, employee_row
            )

            if not is_valid:
                errors.append(f"{emp_id} – {result}")
                continue

            workday_field_values = result

            # 🔹 Merge fields (manual + workday)
            field_values = {}

            for col in DYNAMIC_FIELDS:
                normalized_col = col.strip().lower().replace("_", " ")

                # SIGNATORY handled separately
                if normalized_col == "signatory":
                    continue

                # AUTO FIELD
                if normalized_col == "letter effective date":
                    field_values[col] = datetime.now().strftime("%d-%m-%Y")
                    continue

                hro_value = str(self.selected_hro_row[col]).strip().upper()

                # WORKDAY field
                if hro_value == "WORKDAY":
                    field_values[col] = workday_field_values[col]
                    continue

                # MANUAL (or anything not WORKDAY / NA)
                if col in bulk_df.columns:
                    value = row[col]

                    if pd.isna(value):
                        field_values[col] = ""

                    elif "date" in col.lower():
                        try:
                            field_values[col] = pd.to_datetime(value).strftime("%d-%m-%Y")
                        except:
                            field_values[col] = str(value)

                    else:
                        field_values[col] = str(value)
            
            # ==============================
            # 🔥 AUTO TITLE LOGIC (BULK - FINAL)
            # ==============================

            raw_gender = str(employee_row.get("Gender", "")).strip().lower()

            # normalize gender
            if raw_gender in ["male", "m"]:
                title_value = "Mr."
            elif raw_gender in ["female", "f"]:
                title_value = "Ms./Mrs."
            else:
                title_value = ""

            # 🔥 FORCE INSERT / OVERRIDE (no dependency on existing fields)
            field_values["Ms. / Mr."] = title_value
            field_values["Ms./Mr."] = title_value
            field_values["Title"] = title_value
            field_values["Salutation"] = title_value

            # 🔥 ALSO REPLACE ANY EXISTING MATCHING KEYS
            for key in list(field_values.keys()):
                normalized_key = key.strip().lower().replace("_", "").replace(" ", "")

    def generate_letter(self, win, is_pdf=False):
        
        if not self.current_output_path:
            self.current_output_path = self.OUTPUT_FOLDER
        
        
        # 🚫 Mandatory field validation
        missing_fields = []
        invalid_date_fields = []

        for field, var in self.field_vars.items():
            value = str(var.get()).strip()

            # Empty check
            if not value:
                missing_fields.append(field)
                continue

            # Date format validation
            if "date" in field.lower():
                if not validate_date_format(value):
                    invalid_date_fields.append(field)



        if missing_fields:
            messagebox.showerror(
                "Missing Information",
                "Please fill all mandatory fields:\n\n" + "\n".join(missing_fields)
            )
            return

        if invalid_date_fields:
            messagebox.showerror(
                "Invalid Date Format",
                "The following date fields must be in DD-MM-YYYY format:\n\n"
                + "\n".join(invalid_date_fields)
            )
            return

        if missing_fields:
            messagebox.showerror(
                "Missing Information",
                "Please fill all mandatory fields:\n\n" + "\n".join(missing_fields)
            )
            return
        

        # Close Dynamic Fields screen ONCE
        win.destroy()

        # Open progress window
        progress_win, bar, percent_var = self.show_progress_window("Generating letter... wait till you get confirmation")
        progress_win.update()

        try:
            # 0%
            self.update_progress(progress_win, bar, percent_var, 0)

            template_path = os.path.join(
                self.TEMPLATE_FOLDER,
                f"{self.selected_hro_row[UNIQUE_TEMPLATE_COL]}.docx"
            )

            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")

            run_output_folder = os.path.join(
                self.current_output_path,
                f"Single_Output_{timestamp}"
            )

            entity = str(
                self.selected_hro_row[ENTITY_COL]
            ).strip()

            entity_folder = os.path.join(
                run_output_folder,
                entity
            )

            manager_folder = os.path.join(
                entity_folder,
                self.manager_name.strip()
            )

            os.makedirs(manager_folder, exist_ok=True)

            emp_id = str(self.selected_employee_row[EMP_ID_COL]).strip()

            entity = str(self.selected_hro_row[ENTITY_COL]).strip()
            letter_type = str(self.selected_hro_row[LETTER_COL]).strip()

            emp_name = (
                str(self.selected_employee_row["Forename"]).strip()
                + " "
                + str(self.selected_employee_row["Surname"]).strip()
            )

            def clean_filename(text):
                return re.sub(r'[\\/*?:"<>|]', "", str(text)).strip()

            entity = clean_filename(entity)
            letter_type = clean_filename(letter_type)
            emp_name = clean_filename(emp_name)

            base_name = f"{entity}_{letter_type}_{emp_id}_{emp_name}"

            field_values = {
                k: v.get()
                for k, v in self.field_vars.items()
                if k.strip().lower() != "signatory"
            }

            # ==============================
            # 🔥 AUTO TITLE LOGIC (SINGLE - FINAL)
            # ==============================

            raw_gender = str(self.selected_employee_row.get("Gender", "")).strip().lower()

            # normalize gender
            if raw_gender in ["male", "m"]:
                title_value = "Mr."
            elif raw_gender in ["female", "f"]:
                title_value = "Ms./Mrs."
            else:
                title_value = ""

            # 🔥 FORCE INSERT / OVERRIDE
            field_values["Ms. / Mr."] = title_value
            field_values["Ms./Mr."] = title_value
            field_values["Title"] = title_value
            field_values["Salutation"] = title_value

            # 🔥 ALSO UPDATE EXISTING KEYS
            for key in list(field_values.keys()):
                normalized_key = key.strip().lower().replace("_", "").replace(" ", "")

                if normalized_key in ["Ms. / Mr.", "ms./mr", "title", "salutation"]:
                    field_values[key] = title_value

            self.update_progress(progress_win, bar, percent_var, 20)

            entity = str(
                self.selected_hro_row[ENTITY_COL]
            ).strip()

            entity_folder = os.path.join(
                run_output_folder,
                entity
            )

                entity_folder,
                self.manager_name.strip()

            # ================= WORD ONLY =================
            if not is_pdf:
                word_file = os.path.join(manager_folder, f"{base_name}.docx")

                # 40%
                self.update_progress(progress_win, bar, percent_var, 40)

                replace_in_paragraphs(doc, field_values)
                replace_in_tables(doc, field_values)

                # 🔹 INSERT SIGNATORY IMAGE

                self.update_progress(progress_win, bar, percent_var, 70)

                doc.save(word_file)

                # 100%
                self.update_progress(progress_win, bar, percent_var, 100)

                self.create_log(word_file,
                self.selected_employee_row[EMP_ID_COL],self.manager_name)

                progress_win.destroy()
                messagebox.showinfo(
                    "Success",
                    f"Word file generated:\n\n{word_file}"

            # ================= PDF ONLY =================
            else:
                temp_word = os.path.join(manager_folder, f"{base_name}_temp.docx")
                pdf_path = os.path.join(manager_folder, f"{base_name}.pdf")

                doc = Document(template_path)

                self.update_progress(progress_win, bar, percent_var, 40)

                replace_in_paragraphs(doc, field_values)
                replace_in_tables(doc, field_values)
                # 🔹 INSERT SIGNATORY IMAGE

                    insert_signatory_image(doc, self.signatory_image_path.get())
                
                # 60%
                self.update_progress(progress_win, bar, percent_var, 60)

                doc.save(temp_word)
                gc.collect()
                time.sleep(0.5)
                # 85%
                self.update_progress(progress_win, bar, percent_var, 85)

                # ==============================
                # DETERMINE PDF PASSWORD
                # ==============================

                if self.password_mode.get() == "system":

                    first_name = str(
                        self.selected_employee_row["Forename"]
                    ).strip().replace(" ", "").upper()

                    first_four = first_name[:4] if len(first_name) >= 4 else first_name.ljust(4, "X")

                    dob = self.selected_employee_row[" Date of Birth"]

                    if pd.notna(dob):
                        if isinstance(dob, (pd.Timestamp, datetime)):
                            birth_year = dob.strftime("%Y")
                        else:
                            birth_year = pd.to_datetime(dob).strftime("%Y")
                    else:
                        raise ValueError("Date of Birth is missing for this employee in Workday Input.xlsx")

                    password = first_four + birth_year

                else:
                    password = self.custom_password.get().strip()

                # ==============================
                # CONVERT TO PASSWORD PROTECTED PDF
                # ==============================

                convert_word_to_pdf_with_password(
                    temp_word,
                    pdf_path,
                    password
                )

                if os.path.exists(temp_word):
                    os.remove(temp_word)

                # 100%
                self.update_progress(progress_win, bar, percent_var, 100)

                self.create_log(pdf_path,
                self.selected_employee_row[EMP_ID_COL],self.manager_name)

                progress_win.destroy()
                messagebox.showinfo(
                    "Success",
                    f"PDF generated:\n\n{pdf_path}"
                )

                self.destroy()
                
        except Exception as e:
            progress_win.destroy()
            messagebox.showerror("Error", str(e))

            if os.path.exists(self.LOG_FILE):
                existing_df = pd.read_excel(self.LOG_FILE)
                updated_df = pd.concat([existing_df, new_row_df], ignore_index=True)
            else:
                updated_df = new_row_df

            updated_df.to_excel(self.LOG_FILE, index=False)

        except Exception as e:
            print("Log Error:", e)

    # ==========================
    # ZIP OUTPUT FOLDER
    # ==========================
    def zip_output_folder(self):
        if not self.current_output_path:
            return "No folder selected"

        zip_path = self.current_output_path + ".zip"

        with zipfile.ZipFile(zip_path, 'w', zipfile.ZIP_DEFLATED) as zipf:
            for root, dirs, files in os.walk(self.current_output_path):
                for file in files:
                    file_path = os.path.join(root, file)
                    arcname = os.path.relpath(file_path, self.current_output_path)
                    zipf.write(file_path, arcname)

        return zip_path
